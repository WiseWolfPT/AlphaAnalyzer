```javascript
const RATE_LIMIT_MS = 3000;  // 0.33 stocks/s = 3.3 API calls/s ✅

// Add 429 handler
async function fetchWithRetry(url, options, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    const response = await fetch(url, options);

    if (response.status === 429) {
      const waitTime = Math.pow(2, i) * 30000;  // 30s, 60s, 120s
      console.warn(`⚠️ Rate limit hit - waiting ${waitTime/1000}s before retry ${i+1}/${maxRetries}`);
      await sleep(waitTime);
      continue;
    }

    return response;
  }

  throw new Error('Max retries exceeded after HTTP 429');
}
```

```javascript
// Progress indicator
function updateProgress(current, total, startTime) {
  const elapsed = (Date.now() - startTime) / 1000;
  const rate = current / elapsed;
  const remaining = (total - current) / rate;
  const eta = new Date(Date.now() + remaining * 1000);

  process.stdout.write(`\r✓ ${current}/${total} (${((current/total)*100).toFixed(1)}%) | ` +
    `Rate: ${rate.toFixed(2)} stocks/s | ` +
    `ETA: ${eta.toLocaleTimeString()} | ` +
    `Elapsed: ${Math.floor(elapsed/60)}m ${Math.floor(elapsed%60)}s`);
}

// HTTP 429 counter
let http429Count = 0;
```
