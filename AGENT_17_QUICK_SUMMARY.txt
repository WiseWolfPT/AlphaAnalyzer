5. HELPER FUNCTIONS (priority-stocks-index.ts)
================================================================================

isPriorityStock(symbol: string): boolean
  - Check if stock is in priority list
  - Example: isPriorityStock('AAPL') → true

getPriorityStockRegion(symbol: string): 'US' | 'EU' | 'China' | null
  - Get region for priority stock
  - Example: getPriorityStockRegion('ASML.AS') → 'EU'

getPriorityTier(symbol: string): 1 | 2 | 3 | null
  - Get priority tier (1 = highest)
  - Example: getPriorityTier('AAPL') → 1

getWarmingFrequency(symbol: string): number
  - Get warming frequency in minutes
  - Example: getWarmingFrequency('AAPL') → 30

getPriorityStocksByTier(tier: 1 | 2 | 3): string[]
  - Filter stocks by tier
  - Example: getPriorityStocksByTier(1) → ['AAPL', 'MSFT', ...]

getPriorityStocksByRegion(region: 'US' | 'EU' | 'China'): string[]
  - Filter stocks by region
  - Example: getPriorityStocksByRegion('EU') → ['ASML.AS', 'NESN.SW', ...]

getPriorityStocksStats(): object
  - Get statistics (total, by region, by tier, coverage)
