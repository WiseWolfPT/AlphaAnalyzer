🌐 API ENDPOINTS
------------------------------------------------------------------------

1. GET /api/sectors
   → List all 11 GICS sectors with metadata
   Response: { total, totalStocks, sectors[] }

2. GET /api/sectors/:sectorId
   → Get sector details with industries
   Response: { id, name, stockCount, industries[], topStocks[] }

3. GET /api/sectors/:sectorId/stocks?page=1&limit=50&ivOnly=true
   → Paginated stocks in sector
   Response: { sector, page, limit, total, totalPages, stocks[] }

4. GET /api/sectors/distribution
   → Chart data for visualizations
   Response: { totalStocks, distribution[] }

🔍 SERVICE USAGE (Backend)
------------------------------------------------------------------------

import { gicsSectorService } from '../services/gics-sector-service';

// Initialize (async, call once on server start)
await gicsSectorService.initialize();

// Get sector for stock
const sector = gicsSectorService.getSectorForStock('AAPL');
// → 'Information Technology'

// Get all tech stocks
const techStocks = gicsSectorService.getStocksBySector('Information Technology');
// → ['AAPL', 'MSFT', 'NVDA', ...]

// Get tech stocks with IV
const techStocksWithIV = gicsSectorService.getStocksBySectorWithIV('Information Technology');
// → ['AAPL', 'MSFT', 'NVDA', ...] (filtered)

// Get full stock data
const stockData = gicsSectorService.getStockData('AAPL');
// → { symbol, companyName, sector, industry, exchange, ... }

// Get sector distribution
const distribution = gicsSectorService.getSectorDistribution();
// → { 'Information Technology': 95, 'Financials': 74, ... }
